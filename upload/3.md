# Security Scanner File Selection — Expo + React Native

Actual TypeScript code for the supplied screen. Save as `SecurityScannerFileSelectionScreen.tsx`.

## Install

```bash
npx expo install @expo/vector-icons react-native-safe-area-context expo-document-picker
```

## `SecurityScannerFileSelectionScreen.tsx`

```tsx
import React, { useState } from 'react';
import { View, Text, Pressable, ScrollView, StyleSheet, TextInput, useWindowDimensions, StatusBar } from 'react-native';
import * as DocumentPicker from 'expo-document-picker';
import { SafeAreaView } from 'react-native-safe-area-context';
import { Feather, MaterialCommunityIcons } from '@expo/vector-icons';

const C = {
  navy: '#192D47', blue: '#0878F9', pale: '#E5F0FF', bg: '#F7FAFE',
  text: '#172B49', muted: '#60779A', border: '#E0EAF6', white: '#FFFFFF',
  red: '#FF5559', green: '#20B477',
};

type FileRow = {
  id: string; name: string; kind: string; size: string;
  color: string; icon: React.ComponentProps<typeof MaterialCommunityIcons>['name'];
  tab: string;
};

const FILES: FileRow[] = [
  { id: '1', name: 'Project_Proposal.pdf', kind: 'PDF document', size: '2.4 MB', color: '#F0444A', icon: 'file-pdf-box', tab: 'Recent files' },
  { id: '2', name: 'Screenshot_20251030.png', kind: 'Image', size: '1.8 MB', color: '#147CF5', icon: 'file-image', tab: 'Recent files' },
  { id: '3', name: 'Resume.docx', kind: 'Word document', size: '856 KB', color: '#1877F2', icon: 'file-document', tab: 'Recent files' },
  { id: '4', name: 'Budget_Sheet.xlsx', kind: 'Excel spreadsheet', size: '1.2 MB', color: '#19B875', icon: 'file-excel', tab: 'Recent files' },
  { id: '5', name: 'Source_Code.zip', kind: 'ZIP archive', size: '4.6 MB', color: '#71839E', icon: 'file-archive', tab: 'Recent files' },
  { id: '6', name: 'Meeting_Notes.docx', kind: 'Word document', size: '420 KB', color: '#1877F2', icon: 'file-document', tab: 'Documents' },
  { id: '7', name: 'Project_Plan.pdf', kind: 'PDF document', size: '1.6 MB', color: '#F0444A', icon: 'file-pdf-box', tab: 'Documents' },
  { id: '8', name: 'Screenshot.png', kind: 'Image', size: '1.8 MB', color: '#147CF5', icon: 'file-image', tab: 'Images' },
  { id: '9', name: 'Photo.jpg', kind: 'Image', size: '3.1 MB', color: '#147CF5', icon: 'file-image', tab: 'Images' },
  { id: '10', name: 'Download.zip', kind: 'ZIP archive', size: '4.6 MB', color: '#71839E', icon: 'file-archive', tab: 'Downloads' },
];

const TABS = [
  { label: 'Recent files', icon: 'clock' as const },
  { label: 'Documents', icon: 'file-text' as const },
  { label: 'Images', icon: 'image' as const },
  { label: 'Downloads', icon: 'download' as const },
];

function Sidebar({ onNavigate }: { onNavigate: (page: string) => void }) {
  const items = [
    { label: 'Home', icon: 'home' as const },
    { label: 'History', icon: 'clock' as const },
    { label: 'Quarantine', icon: 'archive' as const },
    { label: 'Settings', icon: 'settings' as const },
  ];
  return (
    <View style={s.sidebar}>
      <View style={s.brand}>
        <MaterialCommunityIcons name="shield-check" size={39} color={C.blue} />
        <Text style={s.brandText}>Security <Text style={{ color: '#8EC5FF' }}>Scanner</Text></Text>
      </View>
      <View style={s.navList}>
        {items.map(item => (
          <Pressable key={item.label} onPress={() => onNavigate(item.label)} style={s.navItem}>
            <Feather name={item.icon} size={21} color="#D4E0F0" />
            <Text style={s.navText}>{item.label}</Text>
            {item.label === 'Quarantine' && <View style={s.badge}><Text style={s.badgeText}>2</Text></View>}
          </Pressable>
        ))}
      </View>
      <View style={s.privacy}>
        <MaterialCommunityIcons name="shield-check-outline" size={26} color="#75B8FF" />
        <View style={{ flex: 1 }}>
          <Text style={s.privacyTitle}>Files stay on your phone by default.</Text>
          <Text style={s.privacyCopy}>We don't upload your files without your permission.</Text>
        </View>
      </View>
    </View>
  );
}

export default function SecurityScannerFileSelectionScreen() {
  const { width } = useWindowDimensions();
  const desktop = width >= 1000;
  const [tab, setTab] = useState('Recent files');
  const [selected, setSelected] = useState<string | null>(null);
  const [deviceFile, setDeviceFile] = useState<DocumentPicker.DocumentPickerAsset | null>(null);
  const [message, setMessage] = useState('');
  const visible = FILES.filter(file => file.tab === tab);
  const chosen = deviceFile?.name ?? visible.find(file => file.id === selected)?.name;

  const browse = async () => {
    try {
      const result = await DocumentPicker.getDocumentAsync({ type: '*/*', copyToCacheDirectory: true });
      if (!result.canceled) {
        setDeviceFile(result.assets[0]);
        setSelected(null);
        setMessage(`Selected ${result.assets[0].name}. Tap “Scan selected file” to continue.`);
      }
    } catch {
      setMessage('Could not open the device file picker. Please try again.');
    }
  };

  const navigate = (page: string) => {
    setMessage(page === 'Home' ? '' : `${page} screen is not included in this component yet.`);
  };

  return (
    <SafeAreaView style={s.safe}>
      <StatusBar barStyle="light-content" backgroundColor={C.navy} />
      <View style={s.app}>
        {desktop && <Sidebar onNavigate={navigate} />}
        <View style={s.content}>
          <View style={s.topbar}>
            {!desktop && <Text style={s.mobileBrand}>🛡 Security <Text style={{ color: '#8EC5FF' }}>Scanner</Text></Text>}
            <Feather name="user" size={20} color={C.navy} />
            <Text style={s.user}>User</Text>
            <Feather name="chevron-down" size={16} color="#8CA0BC" />
          </View>
          <ScrollView contentContainerStyle={[s.page, !desktop && s.pageMobile]}>
            <Pressable onPress={() => navigate('Home')} style={s.back}>
              <Feather name="arrow-left" size={20} color={C.blue} />
              <Text style={s.backText}>Back to Home</Text>
            </Pressable>

            <View style={s.intro}>
              <View style={s.heroIcon}>
                <MaterialCommunityIcons name="file-document-check-outline" size={72} color={C.blue} />
              </View>
              <Text style={s.heading}>Choose a file to scan</Text>
              <Text style={s.subtitle}>Select a file from your device. You choose which file to share{"\n"}with the scanner. Files stay on your phone by default.</Text>
            </View>

            {!!message && <View style={s.message}><Feather name="info" size={17} color={C.muted} /><Text style={s.messageText}>{message}</Text><Pressable onPress={() => setMessage('')}><Feather name="x" size={16} color={C.muted} /></Pressable></View>}

            <View style={s.panel}>
              <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={s.tabs}>
                {TABS.map(item => (
                  <Pressable key={item.label} onPress={() => { setTab(item.label); setSelected(null); setDeviceFile(null); setMessage(''); }} style={[s.tab, tab === item.label && s.tabActive]}>
                    <Feather name={item.icon} size={19} color={tab === item.label ? C.blue : '#8296B5'} />
                    <Text style={[s.tabText, tab === item.label && s.tabTextActive]}>{item.label}</Text>
                  </Pressable>
                ))}
              </ScrollView>

              <View style={s.fileList}>
                {visible.map(file => {
                  const active = selected === file.id;
                  return (
                    <Pressable key={file.id} onPress={() => { setSelected(file.id); setDeviceFile(null); setMessage(''); }} style={[s.fileRow, active && s.fileRowSelected]}>
                      <View style={[s.fileIcon, { backgroundColor: file.color }]}><MaterialCommunityIcons name={file.icon} size={25} color={C.white} /></View>
                      <Text numberOfLines={1} style={s.fileName}>{file.name}</Text>
                      <Text numberOfLines={1} style={s.fileKind}>{file.kind}</Text>
                      <Text style={s.fileSize}>{file.size}</Text>
                      <View style={[s.radio, active && s.radioActive]}>{active && <View style={s.radioDot} />}</View>
                    </Pressable>
                  );
                })}
              </View>

              {!!chosen && <View style={s.chosen}><Feather name="check-circle" size={16} color={C.green} /><Text numberOfLines={1} style={s.chosenText}>Selected: {chosen}</Text></View>}

              <View style={s.actions}>
                <Pressable style={s.scanButton} onPress={() => setMessage(chosen ? `Ready to scan ${chosen}. Connect your scanning engine to perform an actual threat scan.` : 'Select a file from the list or browse your device first.')}>
                  <Text style={s.scanText}>Scan selected file</Text><Feather name="arrow-right" size={19} color={C.white} />
                </Pressable>
                <Pressable style={s.browseButton} onPress={browse}><Feather name="folder" size={17} color={C.blue} /><Text style={s.browseText}>Browse device</Text></Pressable>
                <Pressable style={s.cancelButton} onPress={() => navigate('Home')}><Text style={s.cancelText}>Cancel</Text></Pressable>
              </View>
            </View>

            <View style={s.info}>
              <Feather name="info" size={17} color="#6688B8" />
              <Text style={s.infoText}>If you can’t find your file, make sure you have the right permission to access it. If the file can’t be read, you’ll see an error message.</Text>
            </View>
          </ScrollView>
        </View>
      </View>
    </SafeAreaView>
  );
}
```

## Use it in `App.tsx`

```tsx
import SecurityScannerFileSelectionScreen from './SecurityScannerFileSelectionScreen';

export default function App() {
  return <SecurityScannerFileSelectionScreen />;
}
```

## Notes
- The list contains sample files matching the screenshot.
- **Browse device** opens the native document picker.
- Selecting a row updates the selected-file state.
- **Scan selected file** currently confirms the selection; connect your scanning engine to perform an actual scan.
- Sidebar pages are placeholders until connected to navigation.
