# Security Scanner — Quarantined Files Screen

Expo + React Native + TypeScript screen based on the supplied screenshot.

## Install

```bash
npx expo install @expo/vector-icons react-native-safe-area-context
```

## `QuarantineScreen.tsx`

```tsx
import React, { useMemo, useState } from 'react';
import {
  Alert, Pressable, ScrollView, StatusBar, StyleSheet, Text, TextInput,
  View, useWindowDimensions,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { Feather, MaterialCommunityIcons } from '@expo/vector-icons';

const C = {
  navy: '#192D47', text: '#172B49', blue: '#0878F9', paleBlue: '#EAF3FF',
  bg: '#F6FAFF', white: '#FFFFFF', muted: '#5D759B', border: '#DFEAF9',
  red: '#FA4246', orange: '#F59A23', green: '#1EB578',
};

type QuarantineFile = {
  id: string; name: string; description: string; type: string;
  date: string; time: string; status: 'Threat found' | 'Suspicious';
  reason: string; icon: 'file-pdf-box' | 'file-image'; iconColor: string;
};

const INITIAL_FILES: QuarantineFile[] = [
  {
    id: '1', name: 'Project_Proposal.pdf', description: 'PDF document',
    type: 'PDF', date: 'Oct 30, 2025', time: '10:24 AM',
    status: 'Threat found', reason: 'Malicious content detected.',
    icon: 'file-pdf-box', iconColor: C.red,
  },
  {
    id: '2', name: 'Screenshots_20251030.png', description: 'Image',
    type: 'Image', date: 'Oct 30, 2025', time: '09:17 AM',
    status: 'Suspicious', reason: 'Suspicious patterns detected.',
    icon: 'file-image', iconColor: C.blue,
  },
];

type Props = {
  onBackHome?: () => void;
  onScanFile?: () => void;
  onViewDetails?: (file: QuarantineFile) => void;
};

export default function QuarantineScreen({
  onBackHome, onScanFile, onViewDetails,
}: Props) {
  const { width } = useWindowDimensions();
  const desktop = width >= 950;
  const [files, setFiles] = useState(INITIAL_FILES);
  const [search, setSearch] = useState('');
  const [menuFor, setMenuFor] = useState<string | null>(null);

  const filteredFiles = useMemo(() => {
    const q = search.trim().toLowerCase();
    return q ? files.filter(f =>
      `${f.name} ${f.description} ${f.type} ${f.status} ${f.reason}`.toLowerCase().includes(q)
    ) : files;
  }, [files, search]);

  const details = (file: QuarantineFile) => {
    if (onViewDetails) onViewDetails(file);
    else Alert.alert(file.name, `${file.status}\n${file.reason}\n\nSample data only.`);
  };

  const removeFile = (file: QuarantineFile, action: 'delete' | 'restore') => {
    const title = action === 'delete' ? 'Delete quarantined file?' : 'Restore file?';
    const message = action === 'delete'
      ? `Remove "${file.name}" from this demo list?`
      : `Only restore "${file.name}" if you trust its source and understand the risk.`;
    Alert.alert(title, message, [
      { text: 'Cancel', style: 'cancel' },
      {
        text: action === 'delete' ? 'Delete' : 'Restore',
        style: action === 'delete' ? 'destructive' : 'default',
        onPress: () => {
          setFiles(current => current.filter(f => f.id !== file.id));
          setMenuFor(null);
          if (action === 'restore') {
            Alert.alert('Demo action', 'Removed from this sample list only. No real file was restored.');
          }
        },
      },
    ]);
  };

  return (
    <SafeAreaView style={s.safe}>
      <StatusBar barStyle="light-content" backgroundColor={C.navy} />
      <View style={s.app}>
        {desktop && <View style={s.sidebar}>
          <View style={s.brand}>
            <MaterialCommunityIcons name="shield-check" size={39} color={C.blue} />
            <Text style={s.brandText}>Security <Text style={s.brandAccent}>Scanner</Text></Text>
          </View>
          <View style={s.navList}>
            <Pressable style={s.navItem} onPress={onBackHome}><Feather name="home" size={21} color="#E1EAF6" /><Text style={s.navText}>Home</Text></Pressable>
            <Pressable style={s.navItem}><Feather name="clock" size={21} color="#E1EAF6" /><Text style={s.navText}>History</Text></Pressable>
            <Pressable style={[s.navItem, s.navActive]}><Feather name="archive" size={21} color={C.white} /><Text style={s.navText}>Quarantine</Text><View style={s.badge}><Text style={s.badgeText}>{files.length}</Text></View></Pressable>
            <Pressable style={s.navItem}><Feather name="settings" size={21} color="#E1EAF6" /><Text style={s.navText}>Settings</Text></Pressable>
          </View>
          <View style={s.privacy}>
            <MaterialCommunityIcons name="shield-check-outline" size={26} color="#75B8FF" />
            <View style={{ flex: 1 }}>
              <Text style={s.privacyTitle}>Files stay on your phone by default.</Text>
              <Text style={s.privacyCopy}>We don't upload your files without your permission.</Text>
            </View>
          </View>
        </View>}

        <View style={s.main}>
          <View style={s.topbar}>
            {!desktop && <Text style={s.mobileBrand}>Security <Text style={s.brandAccent}>Scanner</Text></Text>}
            <View style={s.userIcon}><Feather name="user" size={17} color={C.navy} /></View>
            <Text style={s.userText}>User</Text><Feather name="chevron-down" size={16} color="#8CA0BC" />
          </View>

          <ScrollView contentContainerStyle={[s.page, !desktop && s.pageMobile]}>
            <Pressable style={s.backLink} onPress={onBackHome}>
              <Feather name="arrow-left" size={18} color={C.blue} /><Text style={s.backText}>Back to Home</Text>
            </Pressable>

            <View style={s.heading}>
              <View style={s.headingIcon}><Feather name="archive" size={39} color={C.blue} /></View>
              <View style={s.headingCopy}>
                <Text style={s.pageTitle}>Quarantined files</Text>
                <Text style={s.subtitle}>These files are isolated because they may be harmful. Do not open them in the usual way.</Text>
              </View>
            </View>

            <View style={s.infoBanner}>
              <View style={s.infoCircle}><Feather name="info" size={20} color={C.white} /></View>
              <Text style={s.infoText}>Quarantined files are moved to a safe location on your device. They can't be opened or used by other apps until you take an action (e.g. delete or restore).</Text>
            </View>

            <View style={s.listCard}>
              <View style={[s.listHeader, !desktop && s.listHeaderMobile]}>
                <Text style={s.countText}>{files.length} {files.length === 1 ? 'file' : 'files'} quarantined</Text>
                <View style={s.searchBox}>
                  <Feather name="search" size={17} color={C.muted} />
                  <TextInput value={search} onChangeText={setSearch} placeholder="Search files..." placeholderTextColor="#8295B4" style={s.searchInput} accessibilityLabel="Search quarantined files" returnKeyType="search" />
                  {!!search && <Pressable onPress={() => setSearch('')}><Feather name="x-circle" size={17} color={C.muted} /></Pressable>}
                </View>
              </View>

              {desktop && <View style={s.tableHeader}>
                <Text style={[s.th, { flex: 2.15 }]}>File name</Text>
                <Text style={[s.th, { flex: 0.6 }]}>Type</Text>
                <Text style={[s.th, { flex: 1.1 }]}>Date quarantined ↓</Text>
                <Text style={[s.th, { flex: 2.1 }]}>Reason / status</Text>
                <Text style={[s.th, { width: 145, textAlign: 'center' }]}>Actions</Text>
              </View>}

              {filteredFiles.map(file => <View key={file.id} style={[s.fileRow, !desktop && s.fileRowMobile]}>
                <View style={s.fileIdentity}>
                  <View style={[s.fileIcon, { backgroundColor: file.iconColor }]}>
                    <MaterialCommunityIcons name={file.icon} size={28} color={C.white} />
                  </View>
                  <View style={s.fileNameBlock}>
                    <Text style={s.fileName}>{file.name}</Text>
                    <Text style={s.fileDescription}>{file.description}</Text>
                  </View>
                </View>
                {desktop && <Text style={[s.fileType, { flex: 0.6 }]}>{file.type}</Text>}
                <View style={s.dateBlock}><Text style={s.dateText}>{file.date}</Text><Text style={s.timeText}>{file.time}</Text></View>
                <View style={s.statusBlock}>
                  <View style={s.statusTitleRow}>
                    <Feather name={file.status === 'Threat found' ? 'alert-circle' : 'alert-triangle'} size={19} color={file.status === 'Threat found' ? C.red : C.orange} />
                    <Text style={[s.statusTitle, { color: file.status === 'Threat found' ? C.red : '#D77A08' }]}>{file.status}</Text>
                  </View>
                  <Text style={s.reasonText}>{file.reason}</Text>
                </View>
                <View style={[s.rowActions, !desktop && s.rowActionsMobile]}>
                  <Pressable style={s.detailsButton} onPress={() => details(file)}><Text style={s.detailsButtonText}>View details</Text></Pressable>
                  <Pressable style={s.moreButton} accessibilityLabel={`More actions for ${file.name}`} onPress={() => setMenuFor(menuFor === file.id ? null : file.id)}><Feather name="more-vertical" size={19} color={C.muted} /></Pressable>
                </View>
                {menuFor === file.id && <View style={s.actionMenu}>
                  <Pressable style={s.menuItem} onPress={() => removeFile(file, 'restore')}><Feather name="rotate-ccw" size={16} color={C.text} /><Text style={s.menuText}>Restore file</Text></Pressable>
                  <Pressable style={s.menuItem} onPress={() => removeFile(file, 'delete')}><Feather name="trash-2" size={16} color={C.red} /><Text style={[s.menuText, { color: C.red }]}>Delete permanently</Text></Pressable>
                </View>}
              </View>)}

              {filteredFiles.length === 0 && <View style={s.noResults}>
                <Feather name="search" size={28} color={C.muted} />
                <Text style={s.noResultsTitle}>No matching files</Text>
                <Text style={s.noResultsText}>Try another search term or clear the search.</Text>
                {!!search && <Pressable style={s.detailsButton} onPress={() => setSearch('')}><Text style={s.detailsButtonText}>Clear search</Text></Pressable>}
              </View>}

              <View style={s.emptyPanel}>
                <View style={s.emptyIcon}><Feather name="inbox" size={39} color={C.blue} /></View>
                <Text style={s.emptyTitle}>No quarantined files</Text>
                <Text style={s.emptyText}>Files you quarantine will appear here.</Text>
                <Pressable style={s.scanButton} onPress={onScanFile}><Feather name="file-plus" size={17} color={C.blue} /><Text style={s.scanButtonText}>Scan a file</Text></Pressable>
              </View>
            </View>
          </ScrollView>
        </View>
      </View>
    </SafeAreaView>
  );
}
```

## Use in `App.tsx`

```tsx
import QuarantineScreen from './QuarantineScreen';

export default function App() {
  return (
    <QuarantineScreen
      onBackHome={() => {
        // Navigate to your Home screen.
      }}
      onScanFile={() => {
        // Navigate to your Choose a File screen.
      }}
      onViewDetails={(file) => {
        // Navigate to your scan-result screen.
        console.log('Selected quarantined file:', file.id);
      }}
    />
  );
}
```

## Notes

- The two listed files are sample data to match the screenshot.
- Search filters the list. The three-dot menu offers demo restore and delete actions.
- Restore/delete only update the in-memory demo list. Connect these actions to your native quarantine service before using them as real security operations.
- The screenshot shows two rows and an empty-state panel at the same time. This code preserves that visual layout; the empty panel is illustrative.
- The sidebar appears on wider screens and is hidden on narrow screens.
