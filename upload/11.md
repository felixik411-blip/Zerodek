# Security Scanner — Settings Screen

Expo + React Native + TypeScript implementation of the Settings screen shown in the reference image.

## Dependencies

```bash
npx expo install lucide-react-native
npx expo install react-native-svg
```

type NavItemProps = {
  label: string;
  active?: boolean;
  badge?: string;
  icon: React.ReactNode;
  onPress?: () => void;
};

function NavItem({ label, active, badge, icon, onPress }: NavItemProps) {
  return (
    <TouchableOpacity
      accessibilityRole="button"
      onPress={onPress}
      style={[styles.navItem, active && styles.navItemActive]}
    >
      {icon}
      <Text style={styles.navLabel}>{label}</Text>
      {badge ? (
        <View style={styles.badge}>
          <Text style={styles.badgeText}>{badge}</Text>
        </View>
      ) : null}
    </TouchableOpacity>
  );
}

function SectionHeader({
  icon,
  title,
  subtitle,
}: {
  icon: React.ReactNode;
  title: string;
  subtitle: string;
}) {
  return (
    <View style={styles.sectionHeader}>
      <View style={styles.sectionIcon}>{icon}</View>
      <View style={styles.sectionHeaderText}>
        <Text style={styles.sectionTitle}>{title}</Text>
        <Text style={styles.sectionSubtitle}>{subtitle}</Text>
      </View>
      <ChevronRight size={21} color="#7896c8" />
    </View>
  );
}

function SettingsRow({
  label,
  value,
  chevron,
  onPress,
}: {
  label: string;
  value?: string;
  chevron?: boolean;
  onPress?: () => void;
}) {
  return (
    <TouchableOpacity
      accessibilityRole="button"
      onPress={onPress}
      style={styles.settingsRow}
    >
      <Text style={styles.settingsRowLabel}>{label}</Text>
      {value ? <Text style={styles.settingsRowValue}>{value}</Text> : null}
      {chevron ? <ChevronRight size={18} color="#7896c8" /> : null}
    </TouchableOpacity>
  );
}

## Use it

Import the screen into your Expo app:

```tsx
import SettingsScreen from "./SettingsScreen";

export default function App() {
  return <SettingsScreen />;
}
```

> This builds the Settings UI and local notification toggle. Privacy links and the quarantine-management action are visual placeholders; connect them to your app's actual screens and storage logic.
