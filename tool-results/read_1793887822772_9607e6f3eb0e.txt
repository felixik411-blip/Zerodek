   297→type GetStartedState = 'idle' | 'loading' | 'done';
   298→
   299→type SecurityScannerWelcomeScreenProps = {
   300→  /**
   301→   * Optional navigation hook. When provided (the web preview passes the
   302→   * Next.js router), pressing "Get started" navigates to the real next
   303→   * screen. When omitted (the Expo app has no navigator wired yet), the
   304→   * button falls back to the honest loading + confirmation states.
   305→   */
   306→  onGetStarted?: () => void;
   307→  /**
   308→   * Optional navigation hook for the security-overview dashboard band.
   309→   * Receives one of the NavTarget ids ("scan-history", "quarantine", …).
   310→   * When omitted, dashboard navigation falls back to an honest inline
   311→   * notice instead of pretending a route change happened.
   312→   */
   313→  onNavigate?: (target: NavTarget) => void;
   314→};
   315→
   316→export default function SecurityScannerWelcomeScreen({
   317→  onGetStarted,
   318→  onNavigate,
   319→}: SecurityScannerWelcomeScreenProps) {
   320→  const { width } = useWindowDimensions();
   321→  const isWide = width >= 900;
   322→
   323→  // Interactive states for the Get started button (design brief section 10A:
   324→  // "Show clear loading, empty, error, and confirmation states where relevant").
   325→  const [getStartedState, setGetStartedState] = React.useState<GetStartedState>('idle');
   326→  const [buttonFocused, setButtonFocused] = React.useState(false);
   327→  const timerRef = React.useRef<ReturnType<typeof setTimeout> | null>(null);
   328→
   329→  // Security-overview dashboard state: the period filter drives the stat
   330→  // cards and the recent-activity list; the refresh and the protection
   331→  // toggle keep honest local feedback; navNotice is the fallback shown when
   332→  // no onNavigate prop is connected (the Expo app).
   333→  const [period, setPeriod] = React.useState<Period>('today');
   334→  const [refreshing, setRefreshing] = React.useState(false);
   335→  const [lastUpdated, setLastUpdated] = React.useState('2 min ago');
   336→  const [liveProtection, setLiveProtection] = React.useState(true);
   337→  const [navNotice, setNavNotice] = React.useState<string | null>(null);
   338→  const refreshTimerRef = React.useRef<ReturnType<typeof setTimeout> | null>(null);
   339→  const navTimerRef = React.useRef<ReturnType<typeof setTimeout> | null>(null);
   340→
   341→  React.useEffect(() => {
   342→    return () => {
   343→      if (timerRef.current !== null) {
   344→        clearTimeout(timerRef.current);
   345→      }
   346→      if (refreshTimerRef.current !== null) {
   347→        clearTimeout(refreshTimerRef.current);
   348→      }
   349→      if (navTimerRef.current !== null) {
   350→        clearTimeout(navTimerRef.current);
   351→      }
   352→    };
   353→  }, []);
   354→
   355→  const handleGetStarted = () => {
   356→    if (getStartedState === 'loading') {
   357→      return;
   358→    }
   359→    if (onGetStarted) {
   360→      // A real destination is connected: show the loading state, then go.
   361→      setGetStartedState('loading');
   362→      onGetStarted();
   363→      return;
   364→    }
   365→    // TODO: Navigate to your scanner/onboarding route when it exists.
   366→    // Example: navigation.navigate('Scanner');
   367→    // Until that route exists, show an honest loading state followed by a
   368→    // confirmation state — the app must never pretend a step happened.
   369→    setGetStartedState('loading');
   370→    timerRef.current = setTimeout(() => setGetStartedState('done'), 1200);
   371→  };
   372→
   373→  const snapshot = OVERVIEW[period];
   374→
   375→  const handleNavigate = (target: NavTarget) => {
   376→    if (onNavigate) {
   377→      onNavigate(target);
   378→      return;
   379→    }
   380→    // No navigator connected (the Expo app yet): say so instead of
   381→    // pretending the tap went somewhere.
   382→    setNavNotice("Routing isn't connected in this build yet — it works in the web preview.");
   383→    if (navTimerRef.current !== null) {
   384→      clearTimeout(navTimerRef.current);
   385→    }
   386→    navTimerRef.current = setTimeout(() => setNavNotice(null), 2800);
   387→  };
   388→
   389→  const handleRefresh = () => {
   390→    if (refreshing) {
   391→      return;
   392→    }
   393→    setRefreshing(true);
   394→    refreshTimerRef.current = setTimeout(() => {
   395→      setRefreshing(false);
   396→      setLastUpdated('just now');
   397→    }, 900);
   398→  };
   399→
   400→  const handleToggleProtection = () => {
   401→    setLiveProtection((prev) => !prev);
   402→  };
   403→
   404→  return (
   405→    <SafeAreaView style={styles.safeArea}>
   406→      <StatusBar barStyle="light-content" backgroundColor={C.navyDeep} />
   407→
   408→      <View style={styles.appHeader}>
   409→        <View style={styles.brand}>
   410→          <MaterialCommunityIcons name="shield-check" size={42} color={C.blue} />
   411→          <Text style={styles.brandText}>
   412→            Security <Text style={styles.brandAccent}>Scanner</Text>
   413→          </Text>
   414→        </View>
   415→        <Text style={styles.headerTagline}>Safe files. Safer browsing.</Text>
   416→      </View>
   417→
   418→      <ScrollView
   419→        contentContainerStyle={[
   420→          styles.scrollContent,
   421→          isWide ? styles.wideContent : styles.narrowContent,
   422→        ]}
   423→        showsVerticalScrollIndicator={false}
   424→      >
   425→        <View style={[styles.mainLayout, isWide && styles.mainLayoutWide]}>
   426→          <View style={[styles.leftColumn, isWide && styles.leftColumnWide]}>
   427→            <Text style={styles.eyebrow}>WELCOME</Text>
   428→            <Text style={[styles.heroTitle, !isWide && styles.heroTitleNarrow]}>
   429→              Security <Text style={styles.heroAccent}>Scanner</Text>
   430→            </Text>
   431→            <Text style={styles.subtitle}>
   432→              Check files and links for possible threats.
   433→            </Text>
   434→
   435→            <View style={styles.features}>
   436→              <FeatureRow
   437→                icon="file-document-outline"
   438→                title="Scan selected files"
   439→                description="Check files on your device for malware, viruses and other threats."
   440→              />
   441→              <FeatureRow
   442→                icon="link-variant"
   443→                title="Check website links"
   444→                description="Verify links and URLs for malicious content before you open them."
   445→              />
   446→              <FeatureRow
   447→                icon="shield-check-outline"
   448→                title="Review quarantined files"
   449→                description="See and manage files that were flagged as suspicious or dangerous."
   450→              />
   451→            </View>
   452→
   453→            <Pressable
   454→              accessibilityRole="button"
   455→              accessibilityLabel="Get started"
   456→              accessibilityState={{
   457→                busy: getStartedState === 'loading',
   458→                disabled: getStartedState === 'loading',
   459→              }}
   460→              onPress={handleGetStarted}
   461→              onFocus={() => setButtonFocused(true)}
   462→              onBlur={() => setButtonFocused(false)}
   463→              disabled={getStartedState === 'loading'}
   464→              style={({ pressed }) => [
   465→                styles.getStartedButton,
   466→                pressed && styles.getStartedButtonPressed,
   467→                buttonFocused && styles.getStartedButtonFocused,
   468→                getStartedState === 'loading' && styles.getStartedButtonLoading,
   469→              ]}

   504→            )}
   505→          </View>

   507→          <View style={isWide ? styles.rightColumn : styles.mobileIllustration}>

   509→          </View>
   510→        </View>
   511→
   512→        <View style={[styles.dashboard, isWide && styles.dashboardWide]}>

   514→            <View style={styles.dashHeaderText}>
   515→              <Text style={styles.dashTitle}>Security overview</Text>
