   189→/** Case-insensitive search haystack for one topic (title, description, answer). */
   190→function topicHaystack(topic: HelpTopic): string {
   191→  return [
   192→    topic.title,
   193→    topic.description,
   194→    ...topic.answer.map((block) =>
   195→      block.type === 'bullet' ? `${block.label} ${block.text}` : block.text,
   196→    ),
   197→  ]
   198→    .join(' ')
   199→    .toLowerCase();
   200→}
